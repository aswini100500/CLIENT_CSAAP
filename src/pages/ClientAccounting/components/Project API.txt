                    API ID: {selectedUnit.apiId}
                  </span>

                <span className="ml-2 text-xs bg-blue-200 text-blue-800 px-2 py-1 rounded">
                  Local ID: {selectedUnit.id}

              </h3>
              <p className="text-sm text-yellow-600 mt-1">
                All specifications and floor details will be saved specifically for this unit

              <div className="flex flex-wrap gap-3 mt-2">
                {selectedUnit.apiId && (

                    <FaCheck className="mr-1" />
                    API Synced
                  </span>

                {hasSpecifications(selectedUnit) && (

                    Floor Details Added
                  </span>
                )}
                {selectedUnit.isComplete && (

                  </span>

              <button

                  selectedUnit.isComplete
                    ? "bg-gray-100 text-gray-600"
                    : "bg-green-500 hover:bg-green-600 text-white"

        {/* Generate Units Button */}
        <div className="mb-4">
          <button
            onClick={generateUnits}
            disabled={numUnits < 1 || isSaving}

              numUnits < 1 || isSaving
                ? "bg-gray-300 text-gray-500 cursor-not-allowed"

            {isSaving ? (

                <FaSpinner className="animate-spin mr-2" />
                Creating Units...

          {units.length > 0 && (
            <p className="text-sm text-green-600 mt-2">

          )}
        </div>

        <div className="mt-4">
          <label className="block text-sm font-medium text-gray-700 mb-2">

          </label>
          <div className="grid grid-cols-2 md:grid-cols-3 gap-3">
            {FACILITIES.map((facility) => (
              <label key={facility.key} className="inline-flex items-center cursor-pointer">
                <input
                  type="checkbox"
                  checked={facilities[facility.key] || false}

                <span className="ml-2 text-sm text-gray-700">{facility.label}</span>

                  checked={facilities[facility] || false}

                <span className="ml-2 text-sm text-gray-700">{facility}</span>

                  type="button"
                  className="ml-2 text-gray-500 hover:text-red-500"

                  <FaTrash className="text-xs" />
                </button>
              </label>
            ))} 
          </div>
          <div className="mt-3 flex">
            <input
              type="text"
              value={newFacility}

      {(projectType === PROJECT_TYPES.DUPLEX || projectType === PROJECT_TYPES.TRIPLEX) &&
      units.length > 0 ? (
        <div className="flex flex-col space-y-6">
          {/* First Row - Generated Units (full width) */}
          <div className="w-full">
            <div className="bg-gray-50 p-4 md:p-5 rounded-lg border border-gray-200">
              <h2 className="text-lg font-semibold mb-4 text-indigo-700 flex items-center">
                <FaList className="mr-2" />
                Generated Units ({units.length})
                {selectedUnit && (
                  <span className="ml-4 text-sm font-normal text-yellow-600 bg-yellow-100 px-2 py-1 rounded">
                    ✓ {selectedUnit.name} selected
                  </span>

                <button
                  onClick={() => {

                  id="unit-scroll-container"
                  className="flex overflow-x-auto space-x-3 px-10 py-2 scrollbar-hide scroll-smooth"

                  {units.map((unit, idx) => (
                    <div
                      key={unit.id}

                        selectedUnit?.id === unit.id
                          ? "bg-yellow-100 border-yellow-300 shadow-md ring-2 ring-yellow-400"

                          ? "bg-green-50 border-green-200 hover:bg-green-100"

                  <FaChevronRight className="text-gray-600" />
                </button>

              <div className="mt-3 p-3 bg-blue-50 rounded border border-blue-200">

                              {option}

                          <label className="inline-flex items-center cursor-pointer">
                            <input
                              type="radio"
                              name="individualBoundary"
                              checked={mainInfo.individualBoundary === false}
                              onChange={() => handleInputChange("individualBoundary", false)}
                              className="text-indigo-600 focus:ring-indigo-500 rounded"
                            />
                            <span className="ml-2 text-sm text-gray-700">No</span>
                          </label>
                        </div>
                      </div>
                    </div>

                    <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
                      <div>
                        <label className="block text-sm font-medium text-gray-700 mb-2">
                          Land Area (sqft)
                        </label>
                        <input
                          type="number"
                          value={mainInfo.landArea}
                          onChange={(e) => handleInputChange("landArea", e.target.value)}
                          className="w-full border border-gray-300 rounded-md p-3 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                          placeholder="Enter area"
                        />
                      </div>

                      <div>
                        <label className="block text-sm font-medium text-gray-700 mb-2">
                          Total Built-up Area (sqft)
                        </label>
                        <input
                          type="number"
                          value={mainInfo.totalBuiltUpArea}
                          readOnly
                          className="w-full border border-gray-300 bg-gray-100 rounded-md p-3 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                          placeholder="Auto calculated"
                        />
                      </div>

                      <div>
                        <label className="block text-sm font-medium text-gray-700 mb-2">
                          Ground Floor Area (sqft)
                        </label>
                        <input
                          type="number"
                          value={mainInfo.groundFloorArea}
                          onChange={(e) => handleInputChange("groundFloorArea", e.target.value)}
                          className="w-full border border-gray-300 rounded-md p-3 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                          placeholder="Enter area"
                        />
                      </div>

                      <div>
                        <label className="block text-sm font-medium text-gray-700 mb-2">
                          First Floor Area (sqft)

                        <input
                          type="number"
                          value={mainInfo.firstFloorArea}
                          onChange={(e) => handleInputChange("firstFloorArea", e.target.value)}
                          className="w-full border border-gray-300 rounded-md p-3 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                          placeholder="Enter area"
                        />
                      </div>

                      {mainInfo.secondFloorArea !== undefined && (
                        <div>
                          <label className="block text-sm font-medium text-gray-700 mb-2">
                            Second Floor Area (sqft)
                          </label>
                          <input
                            type="number"
                            value={mainInfo.secondFloorArea}
                            onChange={(e) => handleInputChange("secondFloorArea", e.target.value)}
                            className="w-full border border-gray-300 rounded-md p-3 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                            placeholder="Enter area"
                          />
                        </div>
                      )}

                      <div>
                        <label className="block text-sm font-medium text-gray-700 mb-2">
                          Staircase Area (sqft)
                        </label>
                        <input
                          type="number"
                          value={mainInfo.staircaseArea}
                          onChange={(e) => handleInputChange("staircaseArea", e.target.value)}
                          className="w-full border border-gray-300 rounded-md p-3 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                          placeholder="Enter area"
                        />
                      </div>
                    </div>
                  </div>
                </div>

                {/* Additional Information */}
                <div className="bg-white rounded-lg border border-gray-200 p-6">
                  <h3 className="text-lg font-semibold mb-4 text-indigo-700">
                    Additional Information
                  </h3>

                  <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                    <div>
                      <label className="block text-sm font-medium text-gray-700 mb-1">
                        No. of Open Sides
                      </label>
                      <select
                        value={propertyFeatures.openSides}
                        onChange={(e) =>
                          setPropertyFeatures({
                            ...propertyFeatures,
                            openSides: e.target.value,
                          })
                        }
                        className="w-full border border-gray-300 rounded-md p-2 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                      >
                        <option value="">Select</option>
                        <option value="1">1 Side</option>
                        <option value="2">2 Sides</option>
                        <option value="3">3 Sides</option>
                        <option value="4">4 Sides</option>
                      </select>
                    </div>

                    {/* Broker */}
                    <div>
                      <label className="block text-sm font-medium text-gray-700 mb-1">
                        Broker
                      </label>
                      <select
                        value={broker}
                        onChange={(e) => setBroker(e.target.value)}
                        className="w-full border border-gray-300 rounded-md p-2 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                      >
                        <option value="">Select a broker</option>
                        {BROKER_LIST.map((brokerItem) => (
                          <option key={brokerItem.id} value={brokerItem.id}>
                            {brokerItem.name} - {brokerItem.phone}
                          </option>
                        ))}
                      </select>
                    </div>

                    {/* Purchaser */}
                    <div>
                      <label className="block text-sm font-medium text-gray-700 mb-1">
                        Purchaser
                      </label>
                      <input
                        type="text"
                        value={purchaser}
                        onChange={(e) => setPurchaser(e.target.value)}
                        className="w-full border border-gray-300 rounded-md p-2 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                        placeholder="Enter purchaser name"
                      />
                    </div>

                    {/* Staff Engaged */}
                    <div>
                      <label className="block text-sm font-medium text-gray-700 mb-1">
                        Staff Engaged (if any)
                      </label>
                      <input
                        type="text"
                        value={staffEngaged}
                        onChange={(e) => setStaffEngaged(e.target.value)}
                        className="w-full border border-gray-300 rounded-md p-2 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                        placeholder="Enter staff name or ID"
                      />
                    </div>

                    {/* Possession Status */}
                    <div>
                      <label className="block text-sm font-medium text-gray-700 mb-1">
                        Possession Status
                      </label>
                      <select
                        value={possessionStatus}
                        onChange={(e) => setPossessionStatus(e.target.value)}
                        className="w-full border border-gray-300 rounded-md p-2 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                      >
                        <option value="">Select possession status</option>
                        <option value="Under Construction">Under Construction</option>
                        <option value="Ready to Move">Ready to Move</option>
                        <option value="Possession Given">Possession Given</option>
                        <option value="Not Applicable">Not Applicable</option>
                      </select>
                    </div>

                    {/* Loan Provider */}
                    <div>
                      <label className="block text-sm font-medium text-gray-700 mb-1">
                        Loan Provider
                      </label>
                      <input
                        type="text"
                        value={loanProvider}
                        onChange={(e) => setLoanProvider(e.target.value)}
                        className="w-full border border-gray-300 rounded-md p-2 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                        placeholder="Enter loan provider name"
                      />
                    </div>

                    {/* Constructor Name */}
                    <div>
                      <label className="block text-sm font-medium text-gray-700 mb-1">
                        Contractor
                      </label>
                      <input
                        type="text"
                        value={constructor}
                        onChange={(e) => setConstructor(e.target.value)}
                        className="w-full border border-gray-300 rounded-md p-2 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                        placeholder="Enter constructor name"
                      />
                    </div>

                    {/* Constructor Work Type */}
                    <div>
                      <label className="block text-sm font-medium text-gray-700 mb-1">
                        Contractor Work Type
                      </label>
                      <select
                        value={constructorWorkType}
                        onChange={(e) => setConstructorWorkType(e.target.value)}
                        className="w-full border border-gray-300 rounded-md p-2 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500"
                      >
                        <option value="">Select work type</option>
                        <option value="Civil">Civil</option>
                        <option value="Electrical">Electrical</option>
                        <option value="Plumbing">Plumbing</option>
                        <option value="Finishing">Finishing</option>
                        <option value="Structural">Structural</option>
                        <option value="Interior">Interior</option>
                        <option value="Other">Other</option>
                      </select>
                    </div>
                  </div>
                </div>

                {/* Approval Status */}
                <div className="bg-white rounded-lg border border-gray-200 p-6">
                  <h3 className="text-lg font-semibold mb-4 text-indigo-700">
                    Approval Status
                  </h3>
                  <div className="space-y-3">
                    {approvalStatus.map((approval, index) => (
                      <div
                        key={index}
                        className="grid grid-cols-1 md:grid-cols-12 gap-3 items-end bg-gray-50 p-3 rounded-lg border border-gray-200"
                      >
                        <div className="md:col-span-5">
                          <label className="block text-sm font-medium text-gray-700 mb-1">
                            Approval Authority
                          </label>
                          <input
                            type="text"
                            value={approval.authority}
                            onChange={(e) =>
                              handleApprovalChange(index, "authority", e.target.value)

                            className="w-full border border-gray-300 rounded-md p-2 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500 text-sm"
                            placeholder="e.g., RERA, Local Authority"
                          />
                        </div>

                        <div className="md:col-span-5">
                          <label className="block text-sm font-medium text-gray-700 mb-1">
                            Status
                          </label>
                          <select
                            value={approval.status}
                            onChange={(e) =>
                              handleApprovalChange(index, "status", e.target.value)
                            }
                            className="w-full border border-gray-300 rounded-md p-2 focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500 text-sm"
                          >
                            <option value="">Select status</option>
                            <option value="Approved">Approved</option>
                            <option value="Pending">Pending</option>
                            <option value="Rejected">Rejected</option>
                            <option value="Applied">Applied</option>
                          </select>
                        </div>

                        <div className="md:col-span-2 flex justify-center">
                          {index === approvalStatus.length - 1 ? (
                            <button
                              onClick={addApprovalAuthority}
                              className="w-full bg-indigo-500 hover:bg-indigo-600 text-white text-sm py-2 rounded-md transition duration-300 flex items-center justify-center space-x-1"
                            >
                              <span>+ Add</span>
                            </button>
                          ) : (
                            <button
                              onClick={() => removeApprovalAuthority(index)}
                              className="w-full bg-red-500 hover:bg-red-600 text-white text-sm py-2 rounded-md transition duration-300 flex items-center justify-center space-x-1"
                            >
                              <span>Remove</span>
                            </button>
                          )}
                        </div>
                      </div>
                    ))}
                  </div>
                </div>

                {/* Final Save Button for Specifications with API Integration */}
                {showSpecifications && selectedUnit && (
                  <div className="flex justify-end pt-4">
                    <button
                      onClick={handleFinalSave}
                      disabled={isSaving}
                      className="bg-green-600 hover:bg-green-700 text-white px-6 py-3 rounded-md transition duration-300 font-medium flex items-center disabled:bg-green-400 disabled:cursor-not-allowed"
                    >
                      {isSaving ? (
                        <>
                          <FaSpinner className="animate-spin mr-2" />
                          Saving to API...
                        </>
                      ) : (
                        <>
                          <FaCheck className="mr-2" />
                          Save Specifications & Continue to Floors
                        </>
                      )}
                    </button>
                  </div>
                )}
              </div>
            ) : (
              <div className="bg-gray-50 rounded-lg border border-gray-200 p-8 text-center">
                <FaHome className="mx-auto h-12 w-12 text-gray-400 mb-4" />
                <h3 className="text-lg font-semibold text-gray-700 mb-2">
                  No Unit Selected
                </h3>
                <p className="text-gray-500">
                  Please select a unit from the list above to view and edit specifications.
                </p>
              </div>
            )}
          </div>
        </div>
      ) : (
        /* Original layout for other project types or when no units */
        <div>
          {units.length > 0 && (
            <div className="bg-gray-50 p-4 md:p-5 rounded-lg border border-gray-200">
              <h2 className="text-lg font-semibold mb-4 text-indigo-700 flex items-center">
                <FaList className="mr-2" />
                Generated Units ({units.length})
                {selectedUnit && (
                  <span className="ml-4 text-sm font-normal text-yellow-600 bg-yellow-100 px-2 py-1 rounded">
                    ✓ {selectedUnit.name} selected
                  </span>
                )}
              </h2>
              <div className="space-y-3 max-h-80 overflow-y-auto pr-2">
                {units.map((unit, idx) => (
                  <div
                    key={unit.id}
                    className={`relative p-4 border rounded-lg flex items-center cursor-pointer transition-all duration-200 ${
                      selectedUnit?.id === unit.id
                        ? "bg-yellow-100 border-yellow-300 shadow-md"
                        : unit.apiId
                        ? "bg-green-50 border-green-200"
                        : "bg-yellow-50 border-yellow-200"
                    }`}
                    onClick={() => onUnitClick(unit)}
                  >
                    <span
                      className={`font-bold mr-3 ${
                        selectedUnit?.id === unit.id
                          ? "text-yellow-700"
                          : unit.apiId
                          ? "text-green-700"
                          : "text-yellow-700"
                      }`}
                    >
                      {idx + 1}.
                    </span>

                    <div className="flex-1">
                      {editingUnitName === unit.id ? (
                        <div className="flex items-center space-x-2">
                          <input
                            type="text"
                            value={tempUnitName}
                            onChange={(e) => setTempUnitName(e.target.value)}
                            className="border border-gray-300 rounded px-2 py-1 text-sm flex-1"
                            onKeyPress={(e) => e.key === "Enter" && saveUnitName(unit.id)}
                          />
                          <button
                            onClick={() => saveUnitName(unit.id)}
                            className="text-green-600 hover:text-green-800"
                          >
                            <FaCheck className="text-xs" />
                          </button>
                          <button
                            onClick={cancelEditingUnitName}
                            className="text-red-600 hover:text-red-800"
                          >
                            <FaTimes className="text-xs" />
                          </button>
                        </div>
                      ) : (
                        <div className="flex items-center justify-between">
                          <div className="flex items-center">
                            <span
                              className={
                                selectedUnit?.id === unit.id
                                  ? "font-bold text-yellow-800"
                                  : unit.apiId
                                  ? "font-medium text-gray-900"
                                  : "font-semibold text-yellow-800"
                              }
                            >
                              {unit.name}
                            </span>
                            {selectedUnit?.id === unit.id && (
                              <FaArrowRight className="ml-2 text-yellow-600 text-sm" />
                            )}
                          </div>
                          <button
                            onClick={(e) => {
                              e.stopPropagation();
                              startEditingUnitName(unit.id, unit.name);
                            }}
                            className="text-gray-500 hover:text-blue-600 ml-0 mt-4"
                            title="Edit Unit Name"
                          >
                            <FaEdit className="text-xs" />
                          </button>
                        </div>
                      )}

                      {/* Unit Details Summary with Editable Area */}
                      <div className="text-xs text-gray-500 mt-1 flex flex-wrap gap-2 items-center">
                        <EditableArea unit={unit} />

                        {unit.apiId && (
                          <span className="text-green-600 font-medium">✓ API</span>
                        )}
                        {hasSpecifications(unit) && (
                          <span className="text-blue-600 font-medium">✓ Specs</span>
                        )}
                        {hasFloorDetails(unit) && (
                          <span className="text-purple-600 font-medium">✓ Floors</span>
                        )}
                      </div>
                    </div>

                    {unit.apiId ? (
                      <FaCheckCircle className="ml-2 text-green-500 text-sm" />
                    ) : (
                      <FaInfoCircle className="ml-2 text-yellow-500 text-sm" />
                    )}

                    {/* Remove Unit Button */}
                    <button
                      className="absolute top-2 right-2 text-gray-500 p-1 rounded hover:text-red-500"
                      onClick={(e) => {
                        e.stopPropagation();
                        removeUnit(unit.id);
                      }}
                      title="Remove Unit"
                    >
                      <FaTrash className="text-xs" />
                    </button>
                  </div>
                ))}
              </div>

              {/* Unit List Instructions */}
              <div className="mt-3 p-3 bg-blue-50 rounded border border-blue-200">
                <p className="text-sm text-blue-700 flex items-center">
                  <FaInfoCircle className="mr-2" />
                  {selectedUnit
                    ? `Currently editing ${selectedUnit.name}. Click on any unit to switch.`
                    : "Click on any unit to view and edit detailed information"}
                </p>
                {!selectedUnit && units.length > 0 && (
                  <p className="text-xs text-blue-600 mt-1">
                    Select a unit to start adding project specifications and floor details
                  </p>
                )}
                {selectedUnit && !showSpecifications && (
                  <div className="flex justify-end mt-3">
                    <button
                      onClick={handleSaveAndContinue}
                      disabled={isSaving}
                      className="bg-indigo-600 hover:bg-indigo-700 text-white px-6 py-3 rounded-md transition duration-300 font-medium flex items-center disabled:bg-indigo-400 disabled:cursor-not-allowed"
                    >
                      {isSaving ? (
                        <>
                          <FaSpinner className="animate-spin mr-2" />
                          Saving...
                        </>
                      ) : (
                        <>
                          Save & Continue
                          <FaArrowRight className="ml-2" />
                        </>
                      )}
                    </button>
                  </div>
                )}
              </div>
            </div>
          )}
        </div>
      )}
    </div>
  );
};

export default MainInfoSection;
